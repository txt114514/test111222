	private static IFaction GetBesiegerFactionSafe(Settlement settlement)

		{

		}
		catch (Exception ex)
		{

			return false;
		}
		if (mapFaction == null)
		{
			if (_reported.Add(mobileParty.StringId))
			{
				Main.Log("[DefenderGuard] Party with NULL MapFaction inside " + town.Name?.ToString() + ": id=" + mobileParty.StringId + " component=" + ((mobileParty.PartyComponent != null) ? mobileParty.PartyComponent.GetType().Name : "none") + " clan=" + ((mobileParty.ActualClan != null) ? mobileParty.ActualClan.StringId : "null") + " home=" + ((mobileParty.HomeSettlement != null) ? mobileParty.HomeSettlement.StringId : "null"));
			}
			return false;
		}
		if (besiegerFaction != null && !mapFaction.IsAtWarWith(besiegerFaction))
		{
			return false;
		}
		if (!mobileParty.IsActive || mobileParty.IsVillager || mobileParty.IsCaravan)
		{
			return false;
		}

		{
			return false;
		}
		if (battleType == MapEvent.BattleTypes.SallyOut && (mobileParty.PartyComponent is TownSupplyLinePartyComponent || mobileParty.PartyComponent is NavalSupplyConvoyPartyComponent || mobileParty.PartyComponent is CastleForagerPartyComponent))
