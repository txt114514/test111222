	private bool IsCamel(ItemObject item)
	{
		if (item == null || item.StringId == null)
		{
			return false;
		}
		string text = item.StringId.ToLower();
		return text.Contains("camel");
	}

	private float GetTemperatureModifierForHorses(float temperature)
	{
		float result = 1f;
		if (temperature < 5f)
		{
			result = 2f;
		}
		else if (temperature < 15f)
		{
			float num = (15f - temperature) / 10f;
			result = 1f + num * 0.2f;
		}
		else if (temperature > 25f)
		{
			result = 1.5f;
		}
		return result;
	}

	private float GetTemperatureModifierForCamels(float temperature)
	{
		float result = 1f;
		if (temperature < 5f)
		{
			result = 2.5f;
		}
		else if (temperature < 15f)
		{
			float num = (15f - temperature) / 10f;
			result = 1f + num * 0.3f;
		}
		else if (temperature > 30f)
		{
			result = 0.8f;
		}
		return result;
	}

	private float GetCombinedModifierForHorses(MobileParty party, float temperature)
	{
		float terrainModifierForHorses = GetTerrainModifierForHorses(party);
		float temperatureModifierForHorses = GetTemperatureModifierForHorses(temperature);
		return terrainModifierForHorses * temperatureModifierForHorses;
	}

	private float GetCombinedModifierForCamels(MobileParty party, float temperature)
	{
		float terrainModifierForCamels = GetTerrainModifierForCamels(party);
		float temperatureModifierForCamels = GetTemperatureModifierForCamels(temperature);
		return terrainModifierForCamels * temperatureModifierForCamels;
	}

	{
		modifier = 1f;

		{
			return false;
		}
		if (!party.LeaderHero.Culture.StringId.Equals("khuzait", StringComparison.OrdinalIgnoreCase))
		{
			return false;
		}
		if (Campaign.Current == null || Campaign.Current.MapSceneWrapper == null)
		{
			return false;
		}
		TerrainType faceTerrainType = Campaign.Current.MapSceneWrapper.GetFaceTerrainType(party.CurrentNavigationFace);
		if (faceTerrainType == TerrainType.Steppe)
		{
			modifier = 0f;
			return true;
		}
		return false;
	}
