	{
		if (!_materialToWorkshops.TryGetValue(itemId, out var value))
		{
			return false;
		}
		if (_contextRestrictions.TryGetValue(itemId, out var value2) && (value2 & context) == 0)
		{
			return false;
		}
		if (settlement == null)
		{
			return false;
		}
		HashSet<string> workshopTypes = GetWorkshopTypes(settlement);
		if (!value.Overlaps(workshopTypes))
		{
			return false;
		}
		if (CountItemInMarket(settlement, itemId) >= 50)
		{
			return false;
		}
		return true;

	public static bool IsProtectedForCastle(string itemId, Town castle, Settlement tradeBoundTown, ProtectionContext context)
	{
		if (castle != null && FoodSystemConstants.IsEmergencyMode(castle))
		{
			return false;
		}
		if (!_materialToWorkshops.TryGetValue(itemId, out var value))
		{
			return false;
		}
		if (_contextRestrictions.TryGetValue(itemId, out var value2) && (value2 & context) == 0)
		{
			return false;
		}
		if (tradeBoundTown == null)
		{
			return false;
		}
		if (tradeBoundTown.IsUnderSiege)
		{
			return false;
		}
		if (castle?.Owner?.Settlement?.MapFaction != tradeBoundTown.MapFaction)
		{
			return false;
		}
		HashSet<string> workshopTypes = GetWorkshopTypes(tradeBoundTown);
		if (!value.Overlaps(workshopTypes))
		{
			return false;
		}
		if (CountItemInMarket(tradeBoundTown, itemId) >= 50)
		{
			return false;
		}
		return true;
	}

	private static int CountItemInMarket(Settlement settlement, string itemId)
	{
		if (settlement?.ItemRoster == null)
		{
			return 0;
		}
		ItemRoster itemRoster = settlement.ItemRoster;
		for (int i = 0; i < itemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = itemRoster.GetElementCopyAtIndex(i);
			if (elementCopyAtIndex.EquipmentElement.Item != null && string.Equals(elementCopyAtIndex.EquipmentElement.Item.StringId, itemId, StringComparison.OrdinalIgnoreCase))
			{
				return elementCopyAtIndex.Amount;
			}
		}
		return 0;
	}

	private static HashSet<string> GetWorkshopTypes(Settlement settlement)
	{
		string stringId = settlement.StringId;
		if (_workshopCache.TryGetValue(stringId, out var value))
		{
			return value;
		}
		HashSet<string> hashSet = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
		if (settlement.Town != null)
		{
			Workshop[] workshops = settlement.Town.Workshops;
			foreach (Workshop workshop in workshops)
			{
				if (workshop?.WorkshopType != null && !workshop.WorkshopType.IsHidden)
				{
					hashSet.Add(workshop.WorkshopType.StringId);
				}
			}
		}
		_workshopCache[stringId] = hashSet;
		bool flag = false;
		return hashSet;
	}

			return false;

	{
		if (item == null)
		{
			return false;
		}
		return _reservedItemIds.Contains(item.StringId);
	}

	public static void Toggle(ItemObject item)
	{
		if (item != null && CanReserve(item) && !_reservedItemIds.Add(item.StringId))
		{
			_reservedItemIds.Remove(item.StringId);
		}
	}

	public static bool ShouldUseReservedFood(MobileParty party)
	{
		if (party != MobileParty.MainParty)
		{
			return false;
		}
		return GetUnreservedFoodCount(party) <= 0;
	}

	public static int GetUnreservedFoodCount(MobileParty party)
	{
		if (party == null || party.ItemRoster == null)
		{
			return 0;
		}
		if (party != MobileParty.MainParty)
		{
			return GetActualFoodCountFromRoster(party.ItemRoster);
		}
		int num = 0;
		ItemRoster itemRoster = party.ItemRoster;
		for (int i = 0; i < itemRoster.Count; i++)
		{
			ItemObject itemAtIndex = itemRoster.GetItemAtIndex(i);
			if (itemAtIndex != null && itemAtIndex.IsFood && !IsReserved(itemAtIndex))
			{
				num += itemRoster.GetElementNumber(i);
			}
		}
		return num;
	}

	private static int GetActualFoodCountFromRoster(ItemRoster roster)
	{
		int num = 0;
		for (int i = 0; i < roster.Count; i++)
		{
			ItemObject itemAtIndex = roster.GetItemAtIndex(i);
			if (itemAtIndex != null && itemAtIndex.IsFood)
			{
				num += roster.GetElementNumber(i);
			}
		}
		return num;
	}
