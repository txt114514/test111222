	private void OnWarDeclared(IFaction faction1, IFaction faction2, DeclareWarAction.DeclareWarDetail detail)
	{
		try
		{
			foreach (MobileParty item in MobileParty.All)
			{
				if (!TownSupplyLinePartyComponent.IsSupplyLineParty(item))
				{
					continue;
				}
				TownSupplyLinePartyComponent component = TownSupplyLinePartyComponent.GetComponent(item);
				if (component == null || _partiesToDestroy.Contains(item))
				{
					continue;
				}
				bool flag = component.DestinationTown == null || !IsSameFactionAsOrigin(component.DestinationTown, component);
				bool flag2 = component.HomeTown == null || !IsSameFactionAsOrigin(component.HomeTown, component);
				if ((component.State == TownSupplyLinePartyComponent.SupplyLineState.TravelingToDestination) & flag)
				{
					component.State = (flag2 ? TownSupplyLinePartyComponent.SupplyLineState.Fleeing : TownSupplyLinePartyComponent.SupplyLineState.Returning);
					if (!flag2 && component.HomeTown != null)
					{
						component.SetDestination(component.HomeTown);
					}
				}
				else if ((component.State == TownSupplyLinePartyComponent.SupplyLineState.Returning) & flag2)
				{
					component.State = TownSupplyLinePartyComponent.SupplyLineState.Fleeing;
				}
			}
			foreach (MobileParty item2 in MobileParty.All)
			{
				if (!NavalSupplyConvoyPartyComponent.IsNavalSupplyConvoy(item2))
				{
					continue;
				}
				NavalSupplyConvoyPartyComponent component2 = NavalSupplyConvoyPartyComponent.GetComponent(item2);
				if (component2 == null || _partiesToDestroy.Contains(item2))
				{
					continue;
				}
				bool flag3 = component2.DestinationTown == null || !IsSameFactionAsOriginNaval(component2.DestinationTown, component2);
				bool flag4 = component2.HomeTown == null || !IsSameFactionAsOriginNaval(component2.HomeTown, component2);
				if ((component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.TravelingToDestination) & flag3)
				{
					component2.State = (flag4 ? NavalSupplyConvoyPartyComponent.ConvoyState.Fleeing : NavalSupplyConvoyPartyComponent.ConvoyState.Returning);
					if (!flag4 && component2.HomeTown != null)
					{
						component2.SetDestination(component2.HomeTown);
					}
				}
				else if ((component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.Returning) & flag4)
				{
					component2.State = NavalSupplyConvoyPartyComponent.ConvoyState.Fleeing;
				}
			}
		}
		catch (Exception ex)
		{
			Main.Log("ERROR in TownSupplyLineBehavior.OnWarDeclared: " + ex.Message);
		}
	}

	private bool IsSameFactionAsOrigin(Settlement settlement, TownSupplyLinePartyComponent component)
	{
		if (settlement == null || settlement.MapFaction == null)
		{
			return false;
		}
		IFaction faction = component?.OriginalClan?.MapFaction;
		if (faction == null)
		{
			return false;
		}
		return settlement.MapFaction == faction;
	}

	private bool HasArrivedAtSettlement(MobileParty party, Settlement settlement)
	{
		if (party == null || settlement == null)
		{
			return false;
		}
		return party.CurrentSettlement == settlement;
	}

	private int GetSupplyInteraction(MobileParty party)
	{
		if (_interactedSupplyLines.TryGetValue(party, out var value))
		{
			return value;
		}
		return 0;
	}

	private void SetSupplyInteraction(MobileParty party, int interaction)
	{
		if (_interactedSupplyLines.ContainsKey(party))
		{
			_interactedSupplyLines[party] = interaction;
		}
		else
		{
			_interactedSupplyLines.Add(party, interaction);
		}
	}
