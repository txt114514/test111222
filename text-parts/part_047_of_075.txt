	public static ClimateZone GetClimateZoneForVillage(Village village)
	{
		if (_climateCache.TryGetValue(village, out var value))
		{
			return value;
		}
		ClimateZone climateZone = CalculateClimateZoneInternal(village);
		_climateCache[village] = climateZone;
		return climateZone;
	}

	private static ClimateZone CalculateClimateZoneInternal(Village village)
	{
		try
		{
			if (Campaign.Current == null || Campaign.Current.MapSceneWrapper == null)
			{
				return ClimateZone.Temperate;
			}
			Vec2 getPosition2D = village.Settlement.GetPosition2D;
			float num = Campaign.Current.MapSceneWrapper.GetTerrainSize().Y;
			if (num <= 0f)
			{
				num = 1f;
			}
			float num2 = getPosition2D.Y / num;
			float height = 0f;
			Campaign.Current.MapSceneWrapper.GetHeightAtPoint(new CampaignVec2(getPosition2D, isOnLand: true), ref height);
			PathFaceRecord faceIndex = Campaign.Current.MapSceneWrapper.GetFaceIndex(new CampaignVec2(getPosition2D, isOnLand: true));
			TerrainType faceTerrainType = Campaign.Current.MapSceneWrapper.GetFaceTerrainType(faceIndex);
			if (faceTerrainType == TerrainType.Desert || faceTerrainType == TerrainType.Dune)
			{
				return ClimateZone.Desert;
			}
			if (num2 < 0.32f)
			{
				return ClimateZone.Tropical;
			}
			if (num2 > 0.52f || height > 16f)
			{
				return ClimateZone.Cold;
			}
			return ClimateZone.Temperate;
		}
		catch (Exception ex)
		{
			Main.Log($"ERROR calculating climate zone for {village.Settlement.Name}: {ex.Message}");
			return ClimateZone.Temperate;
		}
	}
