	private bool IsAnySupplyConversation()

	private ConversationPartyData GetConversationPartyData()
	{
		ConversationPartyData result = default(ConversationPartyData);
		MobileParty conversationParty = MobileParty.ConversationParty;
		if (conversationParty == null)
		{
			return result;
		}
		TownSupplyLinePartyComponent component = TownSupplyLinePartyComponent.GetComponent(conversationParty);
		if (component != null)
		{
			result.IsValid = true;
			result.IsNaval = false;
			result.HomeTown = component.HomeTown;
			result.DestinationTown = component.DestinationTown;
			result.OriginalClan = component.OriginalClan;
			result.PartyOwner = component.PartyOwner;
			result.OriginFaction = component.OriginalClan?.MapFaction;
			result.IsTraveling = component.State == TownSupplyLinePartyComponent.SupplyLineState.TravelingToDestination || component.State == TownSupplyLinePartyComponent.SupplyLineState.PreparingAtHome;
			result.IsUnloading = component.State == TownSupplyLinePartyComponent.SupplyLineState.UnloadingAtDestination;
			result.IsReturning = component.State == TownSupplyLinePartyComponent.SupplyLineState.Returning;
			result.IsWaitingAtHome = component.State == TownSupplyLinePartyComponent.SupplyLineState.WaitingAtHome;
			result.IsFleeing = component.State == TownSupplyLinePartyComponent.SupplyLineState.Fleeing;
			result.IsDefending = component.State == TownSupplyLinePartyComponent.SupplyLineState.Defending;
			result.IsHoldingOffshore = false;
			result.IsIdle = component.State == TownSupplyLinePartyComponent.SupplyLineState.Idle;
			return result;
		}
		NavalSupplyConvoyPartyComponent component2 = NavalSupplyConvoyPartyComponent.GetComponent(conversationParty);
		if (component2 != null)
		{
			result.IsValid = true;
			result.IsNaval = true;
			result.HomeTown = component2.HomeTown;
			result.DestinationTown = component2.DestinationTown;
			result.OriginalClan = component2.OriginalClan;
			result.PartyOwner = component2.PartyOwner;
			result.OriginFaction = component2.OriginalClan?.MapFaction;
			result.IsTraveling = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.TravelingToDestination || component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.PreparingAtHome;
			result.IsUnloading = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.UnloadingAtDestination;
			result.IsReturning = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.Returning;
			result.IsWaitingAtHome = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.WaitingAtHome;
			result.IsFleeing = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.Fleeing;
			result.IsDefending = false;
			result.IsHoldingOffshore = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.HoldingOffshore;
			result.IsIdle = component2.State == NavalSupplyConvoyPartyComponent.ConvoyState.Idle;
			return result;
		}
		return result;
	}

	private void SetLinkVariables(ConversationPartyData data)
	{
		TextObject text = ((data.HomeTown != null) ? data.HomeTown.EncyclopediaLinkWithName : new TextObject("{=HAP_SupplyFallbackHome}their town"));
		TextObject text2 = ((data.DestinationTown != null) ? data.DestinationTown.EncyclopediaLinkWithName : new TextObject("{=HAP_SupplyFallbackDest}another town"));
		MBTextManager.SetTextVariable("HOME_LINK", text);
		MBTextManager.SetTextVariable("DEST_LINK", text2);
		MBTextManager.SetTextVariable("FACTION_NAME", (data.OriginFaction != null) ? data.OriginFaction.Name : new TextObject("{=HAP_SupplyFallbackFaction}their realm"));
	}

	private void SetFleeTargetVariable(ConversationPartyData data)
	{

	}

	private bool supply_will_surrender_on_condition()
	{
		int num = MobileParty.ConversationParty?.MemberRoster.TotalManCount ?? 0;
		int num2 = MobileParty.MainParty?.MemberRoster.TotalManCount ?? 0;
		if (num <= 0)
		{
			return true;
		}
		float num3 = (float)num2 / (float)num;
		return num3 >= 3f;
	}

	private bool supply_will_fight_on_condition()
	{
		return !supply_will_surrender_on_condition();
	}
