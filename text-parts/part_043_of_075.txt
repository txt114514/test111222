	private static void DetectNavalDLC()
	{
		if (_navalDLCSearched)
		{
			return;
		}
		_navalDLCSearched = true;
		try
		{
			Type type = null;
			Assembly[] assemblies = AppDomain.CurrentDomain.GetAssemblies();
			foreach (Assembly assembly in assemblies)
			{
				type = assembly.GetType("NavalDLC.CharacterDevelopment.NavalPerks");
				if (type != null)
				{
					break;
				}
			}
			if (!(type != null))
			{
				return;
			}
			Type nestedType = type.GetNestedType("Boatswain");
			if (nestedType != null)
			{
				PropertyInfo property = nestedType.GetProperty("SmoothOperator", BindingFlags.Static | BindingFlags.Public);
				if (property != null)
				{
					_smoothOperatorPerk = property.GetValue(null) as PerkObject;
					_navalDLCAvailable = _smoothOperatorPerk != null;
					Main.Log($"War Sails DLC detected: SmoothOperator perk found = {_navalDLCAvailable}");
				}
			}
		}
		catch (Exception ex)
		{
			Main.Log("Naval DLC detection failed (DLC probably not installed): " + ex.Message);
			_navalDLCAvailable = false;
		}
	}

		}
