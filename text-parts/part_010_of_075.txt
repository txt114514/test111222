	private bool forager_not_same_faction_on_condition()
	{
		CastleForagerPartyComponent foragerComponent = CastleForagerPartyComponent.GetForagerComponent(MobileParty.ConversationParty);
		if (foragerComponent?.HomeCastle == null)
		{
			return true;
		}
		Clan ownerClan = foragerComponent.HomeCastle.OwnerClan;
		if (ownerClan == null)
		{
			return true;
		}
		IFaction mapFaction = ownerClan.MapFaction;
		if (mapFaction != null && mapFaction == Hero.MainHero.MapFaction)
		{
			return false;
		}
		return true;
	}

	private bool forager_talk_friendly_on_condition()
	{
		if (MobileParty.ConversationParty == null)
		{
			return false;
		}
		if (!CastleForagerPartyComponent.IsForagerParty(MobileParty.ConversationParty))
		{
			return false;
		}
		CastleForagerPartyComponent foragerComponent = CastleForagerPartyComponent.GetForagerComponent(MobileParty.ConversationParty);
		if (foragerComponent == null)
		{
			return false;
		}
		if (foragerComponent.State == CastleForagerPartyComponent.ForagerState.Defending || foragerComponent.State == CastleForagerPartyComponent.ForagerState.Fleeing)
		{
			return false;
		}
		IFaction faction = foragerComponent.OriginalClan?.MapFaction ?? MobileParty.ConversationParty.MapFaction;
		if (faction == null)
		{
			return false;
		}
		if (faction.IsAtWarWith(Hero.MainHero.MapFaction))
		{
			return false;
		}
		if (GetForagerInteraction(MobileParty.ConversationParty) != 0)
		{
			return false;
		}
		SetForagerInteraction(MobileParty.ConversationParty, 1);
		if (foragerComponent.HomeCastle != null)
		{
			MBTextManager.SetTextVariable("CASTLE_LINK", foragerComponent.HomeCastle.EncyclopediaLinkWithName);
		}
		return true;
	}

	private bool forager_talk_friendly_return_on_condition()
	{
		if (MobileParty.ConversationParty == null)
		{
			return false;
		}
		if (!CastleForagerPartyComponent.IsForagerParty(MobileParty.ConversationParty))
		{
			return false;
		}
		CastleForagerPartyComponent foragerComponent = CastleForagerPartyComponent.GetForagerComponent(MobileParty.ConversationParty);
		if (foragerComponent == null)
		{
			return false;
		}
		if (foragerComponent.State == CastleForagerPartyComponent.ForagerState.Defending || foragerComponent.State == CastleForagerPartyComponent.ForagerState.Fleeing)
		{
			return false;
		}
		IFaction faction = foragerComponent.OriginalClan?.MapFaction ?? MobileParty.ConversationParty.MapFaction;
		if (faction == null || faction.IsAtWarWith(Hero.MainHero.MapFaction))
		{
			return false;

		if (GetForagerInteraction(MobileParty.ConversationParty) != 1)
		{
			return false;

		{
			MBTextManager.SetTextVariable("CASTLE_LINK", foragerComponent.HomeCastle.EncyclopediaLinkWithName);
		}
		return true;
	}
