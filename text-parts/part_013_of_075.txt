	private bool IsFishingPartyEncounter(out bool hasGoods, out bool isEnemy)
	{
		hasGoods = false;
		isEnemy = false;
		if (PlayerEncounter.Current == null)
		{
			return false;
		}
		if (Campaign.Current.CurrentConversationContext != ConversationContext.PartyEncounter)
		{
			return false;
		}
		PartyBase encounteredParty = PlayerEncounter.EncounteredParty;
		if (encounteredParty == null || !encounteredParty.IsMobile)
		{
			return false;
		}
		MobileParty mobileParty = encounteredParty.MobileParty;
		if (!mobileParty.IsVillager || !IsFishingParty(mobileParty))
		{
			return false;
		}
		Settlement homeSettlement = mobileParty.HomeSettlement;
		hasGoods = HasValuableGoods(encounteredParty);
		isEnemy = FactionManager.IsAtWarAgainstFaction(encounteredParty.MapFaction, Hero.MainHero.MapFaction);
		if (homeSettlement != null)
		{
			MBTextManager.SetTextVariable("SETTLEMENT_NAME", homeSettlement.EncyclopediaLinkWithName);
		}
		if (Hero.MainHero.Clan?.Kingdom != null)
		{
			MBTextManager.SetTextVariable("PLAYER_ADDRESS", new TextObject("{=HAP_VillagerAddressLord}my {?PLAYER.GENDER}lady{?}lord{\\?}"));
		}
		else
		{
			MBTextManager.SetTextVariable("PLAYER_ADDRESS", new TextObject("{=HAP_VillagerAddressCommoner}traveler"));
		}
		return true;
	}

	private bool IsFishingParty(MobileParty party)
	{
		if (party == null || party.PartyComponent == null)
		{
			return false;
		}
		string name = party.PartyComponent.GetType().Name;
		return name == "FishingPartyComponent";
	}

	private int GetFisherInteraction(MobileParty party)
	{
		if (_interactedFishers.TryGetValue(party, out var value))
		{
			return value;
		}
		return 0;
	}

	private void SetFisherInteraction(MobileParty party, int interaction)
	{
		if (_interactedFishers.ContainsKey(party))
		{
			_interactedFishers[party] = interaction;
		}
		else
		{
			_interactedFishers.Add(party, interaction);
		}
	}

	private bool HasValuableGoods(PartyBase party)
	{
		if (party?.ItemRoster == null || party.ItemRoster.IsEmpty())
		{
			return false;
		}
		for (int i = 0; i < party.ItemRoster.Count; i++)
		{
			if (party.ItemRoster.GetElementCopyAtIndex(i).EquipmentElement.Item != null && party.ItemRoster.GetElementNumber(i) > 0)
			{
				return true;
			}
		}
		return false;
	}

	private bool fisher_criminal_first_encounter_condition()
	{
		if (!IsFishingPartyEncounter(out var _, out var isEnemy))
		{
			return false;
		}
		if (isEnemy)
		{
			return false;
		}
		MobileParty mobileParty = PlayerEncounter.EncounteredParty.MobileParty;
		if (GetFisherInteraction(mobileParty) != 0)
		{
			return false;
		}
		IFaction mapFaction = PlayerEncounter.EncounteredParty.MapFaction;
		if (mapFaction == null || mapFaction.MainHeroCrimeRating < 3f)
		{
			return false;
		}
		SetFisherInteraction(mobileParty, 1);
		return true;
	}

	private bool fisher_bribe_feasible_condition()
	{
		PartyBase encounteredParty = PlayerEncounter.EncounteredParty;
		if (encounteredParty == null || !encounteredParty.IsMobile)
		{
			return false;
		}
		CalculateFisherBribeAmount(encounteredParty, out var gold, out var items);
		if (gold <= 0 && items.IsEmpty())
		{
			return false;
		}
		int totalManCount = encounteredParty.MobileParty.MemberRoster.TotalManCount;
		int totalManCount2 = MobileParty.MainParty.MemberRoster.TotalManCount;
		if (totalManCount > 0 && (float)totalManCount2 / (float)totalManCount < 1f)
		{
			return false;
		}
		_cachedBribeGold = gold;
		_cachedBribeItems = items;
		bool flag = gold > 0;
		bool flag2 = !items.IsEmpty();
		if (flag & flag2)
		{
			MBTextManager.SetTextVariable("BRIBE_OFFER", new TextObject("{=FisherBribeGoldAndItems}{BRIBE_GOLD}{GOLD_ICON} and {BRIBE_ITEMS}"));
		}
		else if (flag)
		{
			MBTextManager.SetTextVariable("BRIBE_OFFER", new TextObject("{=FisherBribeGoldOnly}{BRIBE_GOLD}{GOLD_ICON}"));
		}
		else
		{
			MBTextManager.SetTextVariable("BRIBE_OFFER", new TextObject("{=FisherBribeItemsOnly}{BRIBE_ITEMS}"));
		}
		MBTextManager.SetTextVariable("BRIBE_GOLD", gold);
		if (flag2)
		{
			bool flag3 = true;
			for (int i = 0; i < items.Count; i++)
			{
				ItemRosterElement elementCopyAtIndex = items.GetElementCopyAtIndex(i);
				if (elementCopyAtIndex.Amount > 0)
				{
					TextObject textObject = new TextObject("{=FisherBribeItemEntry}{AMOUNT} {ITEM_NAME}");
					textObject.SetTextVariable("AMOUNT", elementCopyAtIndex.Amount);
					textObject.SetTextVariable("ITEM_NAME", elementCopyAtIndex.EquipmentElement.Item.Name);
					if (flag3)
					{
						MBTextManager.SetTextVariable("LEFT", textObject);
						flag3 = false;
					}
					else
					{
						MBTextManager.SetTextVariable("RIGHT", textObject);
						MBTextManager.SetTextVariable("LEFT", GameTexts.FindText("str_LEFT_comma_RIGHT").ToString());
					}
				}
			}
			MBTextManager.SetTextVariable("BRIBE_ITEMS", GameTexts.FindText("str_LEFT_ONLY").ToString());
		}
		return true;
	}
