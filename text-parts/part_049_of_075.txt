	private static void InitializeNavalDLC()
	{
		if (_navalDLCInitialized)
		{
			return;
		}
		_navalDLCInitialized = true;
		try
		{
			Type type = AccessTools.TypeByName("NavalDLC.CharacterDevelopment.NavalPerks");
			if (type != null)
			{
				Type nestedType = type.GetNestedType("Boatswain", BindingFlags.Static | BindingFlags.Public);
				if (nestedType != null)
				{
					PropertyInfo property = nestedType.GetProperty("BlessingsOfTheSea", BindingFlags.Static | BindingFlags.Public);
					if (property != null)
					{
						_blessingsOfTheSeaPerk = property.GetValue(null) as PerkObject;
					}
					PropertyInfo property2 = nestedType.GetProperty("PortAuthority", BindingFlags.Static | BindingFlags.Public);
					if (property2 != null)
					{
						_portAuthorityPerk = property2.GetValue(null) as PerkObject;
					}
				}
			}
			Type type2 = AccessTools.TypeByName("NavalDLC.NavalPolicies");
			if (type2 != null)
			{
				PropertyInfo property3 = type2.GetProperty("MaritimeWealEdict", BindingFlags.Static | BindingFlags.Public);
				if (property3 != null)
				{
					_maritimeWealEdictPolicy = property3.GetValue(null) as PolicyObject;
				}
				PropertyInfo property4 = type2.GetProperty("BolsterTheFyrd", BindingFlags.Static | BindingFlags.Public);
				if (property4 != null)
				{
					_bolsterTheFyrdPolicy = property4.GetValue(null) as PolicyObject;
				}
			}
			if (_blessingsOfTheSeaPerk != null || _portAuthorityPerk != null || _maritimeWealEdictPolicy != null || _bolsterTheFyrdPolicy != null)
			{
				Main.Log("Naval DLC integration initialized:");

	private float GetBaseProductionAmount(Village village, string itemId)
	{
		string stringId = village.VillageType.StringId;
		float num = 0f;
		num = ((_specialProductionAmounts.ContainsKey(stringId) && _specialProductionAmounts[stringId].ContainsKey(itemId)) ? _specialProductionAmounts[stringId][itemId] : ((!_baseProductionAmounts.ContainsKey(itemId)) ? 1f : _baseProductionAmounts[itemId]));
		SecondaryProductionType secondaryType = VillageSecondaryProductionBehavior.GetSecondaryType(village);
		if (secondaryType != SecondaryProductionType.None && _secondaryCustomItemAmounts.ContainsKey(secondaryType) && _secondaryCustomItemAmounts[secondaryType].ContainsKey(itemId) && (!_primaryProductionOverrides.ContainsKey(stringId) || !_primaryProductionOverrides[stringId].Contains(itemId)))
		{
			num = _secondaryCustomItemAmounts[secondaryType][itemId];
		}
		if (_primaryProductionOverrides.ContainsKey(stringId) && _primaryProductionOverrides[stringId].Contains(itemId) && secondaryType != SecondaryProductionType.None && _secondaryProductionItems.ContainsKey(secondaryType) && _secondaryProductionItems[secondaryType].Contains(itemId))
		{
			num *= 1.5f;
		}
		if (secondaryType != SecondaryProductionType.None && _secondaryPrimaryOverrides.ContainsKey(secondaryType) && _secondaryPrimaryOverrides[secondaryType].ContainsKey(itemId))
		{
			num = _secondaryPrimaryOverrides[secondaryType][itemId];
		}
		TerrainType villageTerrain = GetVillageTerrain(village);
		if (_terrainProductionMultipliers.ContainsKey(villageTerrain) && _terrainProductionMultipliers[villageTerrain].ContainsKey(itemId))
		{
			num *= _terrainProductionMultipliers[villageTerrain][itemId];
		}
		return num;
	}

	private ClimateZone GetClimateZone(Village village)

	private static TerrainType GetVillageTerrain(Village village)
	{
		try
		{
			if (Campaign.Current == null || Campaign.Current.MapSceneWrapper == null)
			{
				return TerrainType.Plain;
			}
			Vec2 getPosition2D = village.Settlement.GetPosition2D;
			PathFaceRecord faceIndex = Campaign.Current.MapSceneWrapper.GetFaceIndex(new CampaignVec2(getPosition2D, isOnLand: true));
			return Campaign.Current.MapSceneWrapper.GetFaceTerrainType(faceIndex);
		}
		catch (Exception ex)
		{
			Main.Log($"ERROR getting terrain for {village.Name}: {ex.Message}");
			return TerrainType.Plain;
		}
	}

	private float GetClimateMultiplier(ClimateZone climate, Village village, string itemId)
	{
		return climate switch
		{
			ClimateZone.Desert => GetDesertMultiplier(itemId), 
			ClimateZone.Tropical => GetTropicalMultiplier(itemId), 
			ClimateZone.Cold => GetColdMultiplier(itemId), 
			_ => GetTemperateMultiplier(itemId), 
		};
	}

	private float GetClimateSeasonalFactor(ClimateZone climate)
	{
		return climate switch
		{
			ClimateZone.Tropical => 0.75f, 
			ClimateZone.Desert => 0.65f, 
			ClimateZone.Temperate => 1f, 
			ClimateZone.Cold => 1.25f, 
			_ => 1f, 
		};
	}

	private float ApplyClimateSeasonalModifier(float baseSeasonal, ClimateZone climate, bool isDairyOrLivestock)
	{
		if (isDairyOrLivestock)
		{
			return baseSeasonal;
		}
		float climateSeasonalFactor = GetClimateSeasonalFactor(climate);
		float val;
		if (baseSeasonal >= 1f)
		{
			float num = baseSeasonal - 1f;
			val = 1f + num * climateSeasonalFactor;
		}
		else
		{
			float num2 = 1f - baseSeasonal;
			val = 1f - num2 * climateSeasonalFactor;
		}
		val = Math.Max(0f, val);
		return Math.Min(4f, val);
	}

	private bool CanVillageProduceItem(Village village, string itemId)
	{
		string stringId = village.VillageType.StringId;
		if (_primaryProductionOverrides.ContainsKey(stringId) && _primaryProductionOverrides[stringId].Contains(itemId))
		{
			return true;
		}
		if (_universalItems.Contains(itemId))
		{
			return true;
		}
		SecondaryProductionType secondaryType = VillageSecondaryProductionBehavior.GetSecondaryType(village);
		if (secondaryType == SecondaryProductionType.None)
		{
			return false;
		}
		if (_secondaryProductionItems.ContainsKey(secondaryType))
		{
			return _secondaryProductionItems[secondaryType].Contains(itemId);
		}
		return false;
	}

	private HashSet<string> GetEligibleItems(Village village, SecondaryProductionType secondaryType)
	{
		if (secondaryType == SecondaryProductionType.None)
		{
			return new HashSet<string>();
		}
		if (!_secondaryProductionItems.ContainsKey(secondaryType))
		{
			return new HashSet<string>();
		}
		HashSet<string> hashSet = new HashSet<string>(_secondaryProductionItems[secondaryType]);
		if (secondaryType == SecondaryProductionType.Livestock)
		{
			ClimateZone climateZone = GetClimateZone(village);
			if (climateZone == ClimateZone.Tropical || climateZone == ClimateZone.Desert)
			{
				hashSet.Remove("hog");
			}
		}
		return hashSet;
	}

	private float CalculateScalingFactor(Village village, string itemId)
	{
		SecondaryProductionType secondaryType = VillageSecondaryProductionBehavior.GetSecondaryType(village);
		if (secondaryType == SecondaryProductionType.None)
		{
			return 1f;
		}
		HashSet<string> eligibleItems = GetEligibleItems(village, secondaryType);
		if (eligibleItems.Count == 0)
		{
			return 1f;
		}
		int num = eligibleItems.Count;
		int num2 = 0;
		ClimateZone climateZone = GetClimateZone(village);
		CampaignTime.Seasons getSeasonOfYear = CampaignTime.Now.GetSeasonOfYear;
		foreach (string item in eligibleItems)
		{
			if (IsNativeItem(item))
			{
				num--;
				continue;
			}
			float climateMultiplier = GetClimateMultiplier(climateZone, village, item);
			float seasonalMultiplier = GetSeasonalMultiplier(item, climateZone);
			if (climateMultiplier > 0f && seasonalMultiplier > 0f)
			{
				num2++;
			}
		}
		if (num2 == 0 || num == 0)
		{
			return 1f;
		}
		float val = (float)num / (float)num2;
		return Math.Min(val, 3f);
	}

	private bool IsNativeItem(string itemId)
	{
		int result;
		switch (itemId)
		{
		default:
			result = ((itemId == "butter") ? 1 : 0);
			break;
		case "grain":
		case "cow":
		case "sheep":
		case "hog":
		case "cheese":
			result = 1;
			break;
		}
		return (byte)result != 0;
	}

	internal float GetSeasonalMultiplierForSeason(string itemId, ClimateZone climate, CampaignTime.Seasons season)
	{
		bool isDairyOrLivestock = itemId == "butter" || itemId == "cheese";
		float num;
		switch (itemId)
		{
		case "cabbage_vegetable":
			num = GetCabbageSeasonalMultiplier(season);
			break;
		case "leek_vegetable":
			num = GetLeekSeasonalMultiplier(season);
			break;
		case "carrot_vegetable":
			num = GetCarrotSeasonalMultiplier(season);
			break;
		case "melon_fruit":
			num = GetMelonSeasonalMultiplier(season, climate);
			break;
		case "apple_fruit":
			num = GetAppleSeasonalMultiplier(season);
			break;
		case "orange_fruit":
			num = GetOrangeSeasonalMultiplier(season, climate);
			break;
		case "chicken_product":
			num = GetChickenSeasonalMultiplier(season);
			break;
		case "goose_product":
			num = GetGooseSeasonalMultiplier(season);
			break;
		case "egg":
			num = GetEggSeasonalMultiplier(season);
			break;
		case "milk":
			num = GetMilkSeasonalMultiplier(season);
			break;
		case "heavy_cream":
			num = GetHeavyCreamSeasonalMultiplier(season);
			break;
		case "mushroom_vegetable":
			num = GetMushroomSeasonalMultiplier(season);
			break;
		case "honey":
			num = GetHoneySeasonalMultiplier(season, climate);
			break;
		case "nuts":
			num = GetNutsSeasonalMultiplier(season);
			break;
		case "black_pepper":
			num = GetBlackPepperSeasonalMultiplier(season, climate);
			break;
		case "herbs":
			num = GetHerbSeasonalMultiplier(season, climate);
			break;
		case "straw":
			num = GetStrawSeasonalMultiplier(season);
			break;
		case "charcoal":
			num = GetCharcoalSeasonalMultiplier(season);
			break;
		case "cut_stone":
			num = GetCutStoneSeasonalMultiplier(season);
			break;
		case "gold_ore":
			num = GetGoldOreSeasonalMultiplier(season);
			break;
		default:
			return 1f;
		}
		switch (itemId)
		{
		default:

			{
				return ApplyClimateSeasonalModifier(num, climate, isDairyOrLivestock);
			}
			goto case "honey";
		case "honey":
		case "black_pepper":
		case "orange_fruit":
		case "melon_fruit":
			return num;
		}
	}
