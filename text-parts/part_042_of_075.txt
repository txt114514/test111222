			EnsureFoodCategories();

			float maxValue = ((!_foodCategories.Contains(itemCategory)) ? (instance?.TradeGoodPriceFactorCap ?? 10f) : (instance?.FoodPriceFactorCap ?? 5f));
			return MathF.Clamp(value, 0.1f, maxValue);
		}
		return MathF.Clamp(value, 0.8f, 1.3f);
