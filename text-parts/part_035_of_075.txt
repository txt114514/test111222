	private bool IsVillagerPartyEncounter(out bool hasGoods, out bool isEnemy, out bool isHeadingToTown)
	{
		hasGoods = false;
		isEnemy = false;
		isHeadingToTown = false;
		if (PlayerEncounter.Current == null)
		{
			return false;
		}
		if (Campaign.Current.CurrentConversationContext != ConversationContext.PartyEncounter)
		{
			return false;
		}
		PartyBase encounteredParty = PlayerEncounter.EncounteredParty;
		if (encounteredParty == null || !encounteredParty.IsMobile)
		{
			return false;
		}
		MobileParty mobileParty = encounteredParty.MobileParty;
		if (!mobileParty.IsVillager || IsFishingParty(mobileParty) || HerderPartyComponent.IsHerderParty(mobileParty))
		{
			return false;
		}
		Settlement homeSettlement = mobileParty.HomeSettlement;
		hasGoods = HasValuableGoods(encounteredParty);
		isEnemy = FactionManager.IsAtWarAgainstFaction(encounteredParty.MapFaction, Hero.MainHero.MapFaction);
		isHeadingToTown = mobileParty.DefaultBehavior == AiBehavior.GoToSettlement && mobileParty.TargetSettlement != null && mobileParty.TargetSettlement.IsTown;
		if (homeSettlement != null)
		{
			MBTextManager.SetTextVariable("SETTLEMENT_NAME", homeSettlement.EncyclopediaLinkWithName);
		}
		Settlement settlement = null;
		if (homeSettlement?.Village?.TradeBound != null)
		{
			settlement = homeSettlement.Village.TradeBound;
		}
		else if (mobileParty.LastVisitedSettlement != null && mobileParty.LastVisitedSettlement.IsTown)
		{
			settlement = mobileParty.LastVisitedSettlement;
		}
		else if (homeSettlement != null)
		{
			settlement = SettlementHelper.FindNearestTownToSettlement(homeSettlement, mobileParty.NavigationCapability)?.Settlement;
		}
		if (settlement != null)
		{
			MBTextManager.SetTextVariable("TOWN_NAME", settlement.EncyclopediaLinkWithName);
		}
		if (Hero.MainHero.Clan?.Kingdom != null)
		{
			MBTextManager.SetTextVariable("PLAYER_ADDRESS", new TextObject("{=HAP_VillagerAddressLord}my {?PLAYER.GENDER}lady{?}lord{\\?}"));
		}
		else
		{
			MBTextManager.SetTextVariable("PLAYER_ADDRESS", new TextObject("{=HAP_VillagerAddressCommoner}traveler"));
		}
		return true;
	}

	private int GetTransportPackAnimalCount(MobileParty party)
	{
		if (party == null)
		{
			return 0;
		}
		return (int)(0.5f * (float)party.MemberRoster.TotalManCount);
	}

	private int GetSurplusPackAnimalCount(MobileParty party)
	{
		if (party == null)
		{
			return 0;
		}
		int numberOfPackAnimals = party.ItemRoster.NumberOfPackAnimals;
		int transportPackAnimalCount = GetTransportPackAnimalCount(party);
		return MathF.Max(0, numberOfPackAnimals - transportPackAnimalCount);
	}

	private bool IsFishingParty(MobileParty party)
	{
		if (party == null || party.PartyComponent == null)
		{
			return false;
		}
		string name = party.PartyComponent.GetType().Name;
		return name == "FishingPartyComponent";
	}

	private int GetVillagerInteraction(MobileParty party)
	{
		if (_interactedVillagers.TryGetValue(party, out var value))
		{
			return value;
		}
		return 0;
	}

	private void SetVillagerInteraction(MobileParty party, int interaction)
	{
		if (_interactedVillagers.ContainsKey(party))
		{
			_interactedVillagers[party] = interaction;
		}
		else
		{
			_interactedVillagers.Add(party, interaction);
		}
	}

	private bool HasValuableGoods(PartyBase party)
	{
		if (party?.ItemRoster == null || party.ItemRoster.IsEmpty())
		{
			return false;
		}
		int num = (party.IsMobile ? GetSurplusPackAnimalCount(party.MobileParty) : 0);
		for (int i = 0; i < party.ItemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = party.ItemRoster.GetElementCopyAtIndex(i);
			if (elementCopyAtIndex.EquipmentElement.Item == null)
			{
				continue;
			}
			int elementNumber = party.ItemRoster.GetElementNumber(i);
			if (elementNumber > 0)
			{
				if (elementCopyAtIndex.EquipmentElement.Item.ItemCategory != DefaultItemCategories.PackAnimal)
				{
					return true;
				}
				if (num > 0)
				{
					return true;
				}
			}
		}
		return false;
	}

	private bool VillagerWillSurrender()
	{
		PartyBase encounteredParty = PlayerEncounter.EncounteredParty;
		if (encounteredParty == null || !encounteredParty.IsMobile)
		{
			return true;
		}
		int totalManCount = encounteredParty.MobileParty.MemberRoster.TotalManCount;
		int totalManCount2 = MobileParty.MainParty.MemberRoster.TotalManCount;
		if (totalManCount <= 0)
		{
			return true;
		}
		float num = (float)totalManCount2 / (float)totalManCount;
		return num >= 2f;
	}

	private void CalculateVillagerBribeAmount(PartyBase party, out int gold, out ItemRoster items)
	{
		gold = 0;
		items = new ItemRoster();
		if (party?.MobileParty == null)
		{
			return;
		}
		gold = party.MobileParty.PartyTradeGold / 2;
		int surplusPackAnimalCount = GetSurplusPackAnimalCount(party.MobileParty);
		int num = 0;
		for (int i = 0; i < party.ItemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = party.ItemRoster.GetElementCopyAtIndex(i);
			if (elementCopyAtIndex.EquipmentElement.Item == null)
			{
				continue;
			}
			int elementNumber = party.ItemRoster.GetElementNumber(i);
			if (elementNumber <= 0)
			{
				continue;
			}
			if (elementCopyAtIndex.EquipmentElement.Item.ItemCategory == DefaultItemCategories.PackAnimal)
			{
				int num2 = MathF.Max(0, MathF.Min(elementNumber, surplusPackAnimalCount - num));
				if (num2 > 0)
				{
					int num3 = MathF.Max(1, num2 / 2);
					items.AddToCounts(elementCopyAtIndex.EquipmentElement, num3);
					num += num3;
				}
			}
			else
			{
				int number = MathF.Max(1, elementNumber / 2);
				items.AddToCounts(elementCopyAtIndex.EquipmentElement, number);
			}
		}
	}

	private bool villager_criminal_first_encounter_condition()
	{
		if (!IsVillagerPartyEncounter(out var _, out var isEnemy, out var _))
		{
			return false;
		}
		if (isEnemy)
		{
			return false;
		}
		MobileParty mobileParty = PlayerEncounter.EncounteredParty.MobileParty;
		if (GetVillagerInteraction(mobileParty) != 0)
		{
			return false;
		}
		IFaction mapFaction = PlayerEncounter.EncounteredParty.MapFaction;
		if (mapFaction == null || mapFaction.MainHeroCrimeRating <= 3f)
		{
			return false;
		}
		SetVillagerInteraction(mobileParty, 1);
		return true;
	}
