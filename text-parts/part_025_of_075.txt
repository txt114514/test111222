	private float GetProsperityPenalty(MobileParty attackerParty, Settlement settlement, float prosperity)
	{
		int num = attackerParty.MemberRoster?.TotalHealthyCount ?? 0;
		if (attackerParty.Army != null)
		{
			num = attackerParty.Army.TotalHealthyMembers;
		}
		float num2 = -1f * (MathF.Log((float)num * 0.04f + 2f, 2f) * 2.5f + 2.5f) * 0.01f * prosperity;
		return 3f * num2;
	}

	private int GetBuildingLevelsLost(MobileParty attackerParty)
	{
		int num = attackerParty.MemberRoster?.TotalHealthyCount ?? 0;
		if (attackerParty.Army != null)
		{
			num = attackerParty.Army.TotalHealthyMembers;
		}
		int num2 = MathF.Floor(MathF.Log((float)num * 0.02f + 2f, 2f));
		return MathF.Ceiling(3f * (float)num2);
	}

	private int GetArmyGoldGain(MobileParty attackerParty, Settlement settlement, float prosperity)
	{
		float num = -1f * GetProsperityPenalty(attackerParty, settlement, prosperity);
		return MathF.Floor(num * 15f);
	}

	private int GetPartyGoldGain(float totalArmyGold, float partyContribution)
	{
		return MathF.Floor(totalArmyGold * partyContribution / 100f);
	}

	private float GetInfluenceCost(MobileParty attackerParty, Settlement settlement)
	{
		if (attackerParty.Army == null)
		{
			return 0f;
		}
		int num = attackerParty.Army.Parties.Count((MobileParty t) => t != attackerParty && t.LeaderHero != null && t.LeaderHero.GetTraitLevel(DefaultTraits.Mercy) > 0);
		return (3000f + settlement.Town.Prosperity) / 400f * (float)num * 1.25f;
	}

	private static bool IsFoodItem(ItemObject item)
	{
		if (item == null)
		{
			return false;
		}
		if (item.ItemCategory != null && item.ItemCategory.Properties == ItemCategory.Property.BonusToFoodStores)
		{
			return true;
		}
		if (item.IsFood)
		{
			return true;
		}
		return false;
	}
