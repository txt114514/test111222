	protected override void OnBeforeInitialModuleScreenSetAsRoot()
	{
		base.OnBeforeInitialModuleScreenSetAsRoot();
		if (_warSailsInitialized)
		{
			return;
		}
		try
		{
			Type type = null;
			Assembly[] assemblies = AppDomain.CurrentDomain.GetAssemblies();
			foreach (Assembly assembly in assemblies)
			{
				type = assembly.GetType("NavalDLC.ViewModelCollection.Map.NavalSettlementMenuOverlayVM");
				if (type != null)
				{
					break;
				}
			}
			if (type != null)
			{
				string location = typeof(Main).Assembly.Location;
				string directoryName = Path.GetDirectoryName(location);
				string text = Path.Combine(directoryName, "HarvestAndProduction.WarSails.dll");
				if (File.Exists(text))
				{
					Assembly assembly2 = Assembly.LoadFrom(text);
					assembly2.GetType("HarvestAndProduction.WarSails.WarSailsIntegration")?.GetMethod("Initialize")?.Invoke(null, null);
					Log("Production efficiency icon enabled (War Sails DLC detected).");
				}
				else
				{
					Log("WARNING: War Sails detected but companion DLL not found at: " + text);
				}
			}
			else
			{
				Log("War Sails DLC not detected - production efficiency icon disabled.");
			}
			_warSailsInitialized = true;
		}
		catch (Exception ex)
		{
			Log("War Sails integration skipped: " + ex.Message);
		}
	}

		{
