	private void ProcessCastleStashFood(Town castle)

		if (list.Count == 0)

			return;
		}
		list.Sort((StashClaimableItem stashClaimableItem, StashClaimableItem b) => b.SpoilageRate.CompareTo(stashClaimableItem.SpoilageRate));

			if (num3 >= num2)
			{
				break;
			}
			float a = item2.Amount * item2.FoodValue;
			float num4 = MathF.Min(a, num2 - num3);
			int num5 = (int)(num4 / (float)item2.FoodValue);
			if (num5 > 0)
			{
				num5 = Math.Min(num5, item2.Amount);
				float num6 = num5 * item2.FoodValue;
				stash.AddToCounts(item2.Item, -num5);

				num3 += num6;

		if (castle.FoodStocks > (float)num)
		{

		}
		if (num3 > 0f)

			string stringId = castle.Settlement.StringId;

			{
				_dailyStashLoyaltyBoost[stringId] = num7;

	private int GetStashItemFoodValue(ItemObject item)
	{
		if (item == null)
		{
			return 0;
		}
		if (item.IsFood || item.StringId.StartsWith("meat_"))
		{
			return 1;
		}
		string stringId = item.StringId;
		if (stringId == "cow")
		{
			return 4;
		}
		if (stringId == "sheep" || stringId == "hog")
		{
			return 2;
		}
		if (stringId == "chicken_product" || stringId == "goose_product" || stringId == "hare")
		{
			return 1;
		}
		if (item.ItemCategory != null && item.ItemCategory.Properties == ItemCategory.Property.BonusToFoodStores)
		{
			return 1;
		}
		return 0;
	}

	private List<ClaimableItem> GetClaimableFoodItems(Town town, bool isEmergency)
	{
		List<ClaimableItem> list = new List<ClaimableItem>();
		ItemRoster itemRoster = town.Settlement.ItemRoster;
		for (int i = 0; i < itemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = itemRoster.GetElementCopyAtIndex(i);
			ItemObject item = elementCopyAtIndex.EquipmentElement.Item;
			if (item != null && elementCopyAtIndex.Amount > 0 && IsClaimableFood(item, isEmergency, town.Settlement))
			{
				float spoilageRate = GetSpoilageRate(item);
				list.Add(new ClaimableItem
				{
					Item = item,
					Count = elementCopyAtIndex.Amount,
					SpoilageRate = spoilageRate
				});
			}
		}
		return list;
	}

	private bool IsClaimableFood(ItemObject item, bool isEmergency, Settlement settlement)
	{
		if (item == null)
		{
			return false;
		}
		bool flag = item.ItemCategory != null && item.ItemCategory.Properties == ItemCategory.Property.BonusToFoodStores;
		bool isFood = item.IsFood;
		if (!isEmergency)
		{
			return flag && !RawMaterialRegistry.IsProtected(item.StringId, settlement, RawMaterialRegistry.ProtectionContext.MarketClaiming);
		}
		return flag | isFood;
	}

	private bool IsProlongedStarvation(Town town)
	{
		if (town == null || town.Settlement == null)
		{
			return false;
		}
		float starvationDays = StarvationDesertionBehavior.GetStarvationDays(town.Settlement);
		return starvationDays >= 7f;
	}

	{
		if (item == null)
		{
			return 1f;
		}
		if (SpoilageRates.TryGetValue(item.StringId, out var value))
		{
			return value;
		}
		return 1f;
	}
