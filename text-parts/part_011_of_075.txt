			if (foragerComponent != null)

				switch (foragerComponent.State)
				{
				case CastleForagerPartyComponent.ForagerState.TravelingToTarget:
					if (foragerComponent.TargetSettlement != null)
					{
						foragerComponent.SetDestination(foragerComponent.TargetSettlement);
					}
					break;
				case CastleForagerPartyComponent.ForagerState.Returning:
					if (foragerComponent.HomeCastle != null)
					{
						foragerComponent.SetDestination(foragerComponent.HomeCastle);
					}
					break;
				case CastleForagerPartyComponent.ForagerState.Fleeing:
					if (foragerComponent.TargetSettlement != null)
					{
						foragerComponent.SetDestination(foragerComponent.TargetSettlement);
					}
					break;
				}
			}
		}
		PlayerEncounter.LeaveEncounter = true;
	}

	public static Dictionary<string, int> DailySupplyTotals { get; private set; } = new Dictionary<string, int>();

	private static void ResetDailyTotalsIfNeeded()
	{
		int num = (int)CampaignTime.Now.ToDays;
		if (num != _lastResetDay)
		{
			DailySupplyTotals.Clear();
			_lastResetDay = num;
		}
	}

	private static void RecordDailySupply(Settlement castle, int amount)
	{
		ResetDailyTotalsIfNeeded();
		string stringId = castle.StringId;
		if (DailySupplyTotals.ContainsKey(stringId))
		{
			DailySupplyTotals[stringId] += amount;
		}
		else
		{
			DailySupplyTotals[stringId] = amount;
		}
	}

	public static int GetDailySupplyTotal(Settlement castle)
	{
		if (castle == null || DailySupplyTotals == null)
		{
			return 0;
		}
		ResetDailyTotalsIfNeeded();
		int value;
		return DailySupplyTotals.TryGetValue(castle.StringId, out value) ? value : 0;
	}

	public override void RegisterEvents()

	}
