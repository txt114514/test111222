			string stringId = village.VillageType.StringId;
			VillageProductionModel.ClimateZone climateZoneForVillage = VillageProductionModel.GetClimateZoneForVillage(village);
			List<SecondaryProductionType> validSecondaryTypes = GetValidSecondaryTypes(stringId, climateZoneForVillage);
			if (validSecondaryTypes.Count == 0)

		{
			Main.Log("ERROR determining secondary type for village: " + ex.Message);
			return SecondaryProductionType.None;

	private List<SecondaryProductionType> GetValidSecondaryTypes(string primaryType, VillageProductionModel.ClimateZone climate)
	{
		List<SecondaryProductionType> list = new List<SecondaryProductionType>();
		bool flag = primaryType == "cattle_farm" || primaryType == "sheep_farm" || primaryType == "swine_farm";
		int num;
		switch (primaryType)
		{
		default:
			num = ((primaryType == "desert_horse_ranch") ? 1 : 0);

		case "vlandian_horse_ranch":
		case "battanian_horse_ranch":
		case "sturgian_horse_ranch":
		case "steppe_horse_ranch":
			num = 1;
			break;
		}
		bool flag2 = (byte)num != 0;
		bool flag3 = primaryType == "wheat_farm";
		bool flag4 = climate != VillageProductionModel.ClimateZone.Tropical && climate != VillageProductionModel.ClimateZone.Desert;
		list.Add(SecondaryProductionType.Orchard);
		list.Add(SecondaryProductionType.Garden);
		list.Add(SecondaryProductionType.Poultry);
		if (!flag3)
		{
			list.Add(SecondaryProductionType.Farmland);
		}
		if (!flag)
		{
			list.Add(SecondaryProductionType.Livestock);
		}
		if (primaryType == "trapper" || primaryType == "lumberjack")
		{
			list.Add(SecondaryProductionType.Forester);
		}
		switch (primaryType)
		{
		default:

		case "wheat_farm":

	public static SecondaryProductionType GetSecondaryType(Village village)
	{
		if (village == null)
		{
			Main.Log("WARNING: GetSecondaryType called with null village");
			return SecondaryProductionType.None;
		}
		if (village.VillageType == null)
		{
			Main.Log($"WARNING: Village {village.Name} has null VillageType in GetSecondaryType");
			return SecondaryProductionType.None;
		}
		if (!_isSystemReady)
		{
			Main.Log($"WARNING: Secondary production system not ready for {village.Name}");
			return SecondaryProductionType.None;
		}
		if (_secondaryTypeCache.TryGetValue(village, out var value))
		{
			return value;
		}
		Main.Log($"INFO: Village {village.Name} not found in secondary type cache");
		return SecondaryProductionType.None;
	}
