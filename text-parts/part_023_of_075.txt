	private void OnDailyTickTown(Town town)
	{
		if (town == null || town.Settlement == null)
		{
			return;
		}
		if (town.IsCastle)
		{
			if (town.IsUnderSiege)
			{
				Campaign.Current.GetCampaignBehavior<LordCastleResupplyBehavior>()?.ProcessSiegeRations(town);
			}
			return;
		}
		try
		{
			if (FoodSystemConstants.IsEmergencyMode(town))
			{
				ProcessEmergencyFoodClaim(town);
				ProcessEmergencyLivestockSlaughter(town, 1);
				if (town.IsUnderSiege ? (town.FoodStocks + town.FoodChange <= 0f) : IsProlongedStarvation(town))
				{
					ProcessEmergencyLivestockSlaughter(town);
				}
			}
			if (town.IsUnderSiege)
			{
				Campaign.Current.GetCampaignBehavior<LordCastleResupplyBehavior>()?.ProcessSiegeRations(town);
			}
			float starvationDays = StarvationDesertionBehavior.GetStarvationDays(town.Settlement);
			if (starvationDays < 3f)
			{
				_slaughterNotified.Remove(town.Settlement.StringId);
			}
		}
		catch (Exception ex)
		{
			Main.Log($"ERROR in TownFoodEconomyBehavior.OnDailyTickTown for {town.Name}: {ex.Message}");
		}
	}
