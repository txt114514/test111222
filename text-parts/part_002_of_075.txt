	private void OnAiHourlyTick(MobileParty party, PartyThinkParams p)
	{
		try
		{
			if (party == null || p == null || !ShouldProcessParty(party))

			}
			RefreshSettlementCacheIfNeeded();
			bool flag = ShouldDebugParty(party);
			FoodSituation foodSituation = AnalyzeFoodSituation(party);
			if (!foodSituation.NeedsFood)
			{
				if (flag)
				{
					Main.Log($"[AI FOOD DEBUG] === {party.Name} ===");
					Main.Log($"[AI FOOD DEBUG]   SKIPPED: Has {foodSituation.DaysOfFood:F1} days of food (threshold: {5f} days)");
					Main.Log($"[AI FOOD DEBUG]   Food: {foodSituation.TotalFood:F1} | Consumption: {foodSituation.DailyConsumption:F1}/day");
				}
				return;
			}
			if (flag)
			{
				DebugLogFoodSituation(party, foodSituation);
				_debugScoredSettlements = new List<DebugSettlementScore>();
			}
			EvaluateAndScoreSettlements(party, p, foodSituation, flag);
			EvaluateCastlesForResupply(party, p, foodSituation, flag);
			if (flag)
			{
				DebugLogTopSettlements(party);
				_debugScoredSettlements = null;
			}

		catch (Exception ex)
		{
			Main.Log("[AI FOOD] Error in OnAiHourlyTick: " + ex.Message);
		}
	}

	private FoodSituation AnalyzeFoodSituation(MobileParty party)
	{
		FoodSituation result = new FoodSituation
		{
			TotalFood = party.Food,
			DailyConsumption = Math.Max(0.1f, 0f - party.FoodChange),
			TotalPartySize = (party.MemberRoster?.TotalManCount ?? 0),
			PartySpeed = Math.Max(1f, party.Speed),
			IsStarving = (party.Party?.IsStarving ?? false)
		};
		if (party.Army != null && party.Army.LeaderParty == party)
		{
			foreach (MobileParty attachedParty in party.AttachedParties)
			{
				if (attachedParty != null)
				{
					result.TotalFood += attachedParty.Food;
					result.DailyConsumption += Math.Max(0f, 0f - attachedParty.FoodChange);
					result.TotalPartySize += attachedParty.MemberRoster?.TotalManCount ?? 0;
					PartyBase party2 = attachedParty.Party;
					if (party2 != null && party2.IsStarving)
					{
						result.IsStarving = true;
					}
				}
			}
		}
		result.DaysOfFood = ((result.DailyConsumption > 0f) ? (result.TotalFood / result.DailyConsumption) : float.MaxValue);
		float num = 10f - result.DaysOfFood;
		result.FoodNeeded = Math.Max(0f, num * result.DailyConsumption);
		if (result.IsStarving)
		{
			result.UrgencyMultiplier = 9f;
			result.NeedsFood = true;
		}
		else if (result.DaysOfFood < 2f)
		{
			result.UrgencyMultiplier = 3f;
			result.NeedsFood = true;
		}
		else if (result.DaysOfFood < 5f)
		{
			float num2 = (5f - result.DaysOfFood) / 3f;
			result.UrgencyMultiplier = 1f + num2 * 2f;
			result.NeedsFood = true;
		}
		else
		{
			result.UrgencyMultiplier = 1f;
			result.NeedsFood = false;
		}
		return result;
	}
