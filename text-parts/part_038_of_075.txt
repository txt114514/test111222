	public static int GetFoodQualityTier(ItemObject item)
	{
		if (item == null)
		{
			return 1;
		}
		try
		{
			if (item.ItemComponent is TradeItemComponent { MoraleBonus: >0, MoraleBonus: var moraleBonus })
			{
				return moraleBonus;
			}
		}
		catch
		{
		}
		return GetFallbackFoodQualityTier(item.StringId);
	}

	{

			3 => 0.5f, 
			4 => 0.8f, 
			5 => 1f, 
			_ => 0f, 
		};
	}
