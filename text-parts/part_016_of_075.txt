	private void OnDailySettlementTick(Settlement settlement)
	{
		try
		{
			Village village = settlement.Village;
			if (village == null)
			{
				return;
			}
			HAPSettings? instance = GlobalSettings<HAPSettings>.Instance;
			if (instance == null || instance.EnableHerderParties)
			{
				if (HerderPartyComponent.HasActiveHerder(village))
				{
					CheckDisbandConditions(village);
				}
				else if (CanSpawnHerder(village) && !(MBRandom.RandomFloat > 0.5f))
				{
					SpawnHerder(village);
				}
			}
		}
		catch (Exception ex)
		{
			Main.Log("ERROR in HerderBehavior.OnDailySettlementTick: " + ex.Message);
		}
	}

	private bool CanSpawnHerder(Village village)
	{
		if (_shelterDisbandTimes.TryGetValue(village, out var value))
		{
			if (value.ElapsedHoursUntilNow < 48f)
			{
				return false;
			}
			_shelterDisbandTimes.Remove(village);
		}
		string text = village.VillageType?.StringId;
		if (text != "cattle_farm" && text != "sheep_farm" && text != "swine_farm")
		{
			return false;
		}
		if (village.VillageState != Village.VillageStates.Normal)
		{
			return false;
		}
		if (village.Settlement.Party.MapEvent != null)
		{
			return false;
		}
		if (village.Hearth < 300f)
		{
			return false;
		}
		if (village.Bound?.Town == null)
		{
			return false;
		}
		if (village.Bound.Town.Security < 60f)
		{
			return false;
		}
		string animalId = ((text == "cattle_farm") ? "cow" : ((text == "swine_farm") ? "hog" : "sheep"));
		int animalCountInVillage = GetAnimalCountInVillage(village, animalId);
		if (animalCountInVillage < 20)
		{
			return false;
		}
		if (IsSnowCovered(village.Settlement))
		{
			return false;
		}
		CampaignTime.Seasons getSeasonOfYear = CampaignTime.Now.GetSeasonOfYear;
		if (getSeasonOfYear == CampaignTime.Seasons.Winter)
		{
			return false;
		}
		return true;
	}

			return;
		}
		bool flag = false;
		string text = "";
		if (village.Hearth < 300f)
		{
			flag = true;
			text = "low hearth";
		}
		else if (village.Bound?.Town != null && village.Bound.Town.Security < 60f)
		{
			flag = true;
			text = "low security";
		}
		else if (village.VillageState != Village.VillageStates.Normal)
		{
			flag = true;
			text = "village state " + village.VillageState;
		}
		else if (IsSnowCovered(village.Settlement))
		{
			flag = true;
			text = "snow";
		}
		else if (CampaignTime.Now.GetSeasonOfYear == CampaignTime.Seasons.Winter)

					list.Add(item.Character.HeroObject);

	private void OnHourlyTickParty(MobileParty party)
	{
		HerderPartyComponent component;
		if (party.MapEvent != null || (component = HerderPartyComponent.GetComponent(party)) == null)
		{
			return;
		}
		try
		{
			Settlement currentSettlement = party.CurrentSettlement;
			if (currentSettlement != null && component.Village?.Settlement == currentSettlement && party.IsFleeing())
			{
				OnDropOff(component);
				return;
			}
			if (currentSettlement != null)
			{
				Settlement settlement = component.Village?.Settlement;
				if (settlement == null)
				{
					ForceDisbandHerder(component, currentSettlement);
				}
				else if (currentSettlement == settlement)
				{
					if (component.State == HerderPartyComponent.HerderState.ReturningToVillage)
					{
						if (ShouldDisbandNow(component))
						{
							try
							{
								LeaveSettlementAction.ApplyForParty(party);
							}
							catch
							{
							}
							DisbandHerder(component);
						}
						else
						{
							ProcessHerdTurnover(component);
							OnDropOff(component);
							StartRoaming(component);
							bool flag = false;
						}
					}
					else
					{
						try
						{
							LeaveSettlementAction.ApplyForParty(party);
						}
						catch
						{
						}
						_shelterDisbandTimes[component.Village] = CampaignTime.Now;
						bool flag2 = false;
						DisbandHerder(component);
					}
				}
				else
				{
					_shelterDisbandTimes[component.Village] = CampaignTime.Now;
					bool flag3 = false;
					ForceDisbandHerder(component, currentSettlement);
				}
				return;
			}
			if (party.ShortTermBehavior != party.DefaultBehavior || party.IsFleeing())
			{
				MobileParty threatParty = party.ShortTermTargetParty ?? FindNearestHostileParty(party);
				component.ThreatParty = threatParty;
			}
			else if (component.ThreatParty != null)
			{
				component.ThreatParty = null;
			}
			float elapsedHoursUntilNow = component.RoamingStartTime.ElapsedHoursUntilNow;
			if (elapsedHoursUntilNow > 120f)
			{
				if (component.State != HerderPartyComponent.HerderState.ReturningToVillage)
				{
					bool flag4 = false;
					StartReturnToVillage(component);
					return;
				}
				if (elapsedHoursUntilNow > 240f)
				{
					bool flag5 = false;
					DisbandHerder(component);
					return;
				}
			}
			if (party.ShortTermBehavior != party.DefaultBehavior)
			{
				if (party.IsFleeing() && party.ShortTermTargetParty != null && party.ShortTermTargetParty.Position.DistanceSquared(party.TargetPosition) < 16f)
				{
					GoToNewGrazingPoint(component);
				}
			}
			else if (party.DefaultBehavior == AiBehavior.Hold)
			{
				if (component.State == HerderPartyComponent.HerderState.ReturningToVillage)
				{
					StartReturnToVillage(component);
					return;
				}
				if (component.State != HerderPartyComponent.HerderState.Roaming && component.State != HerderPartyComponent.HerderState.Herding)
				{
					StartRoaming(component);
					return;
				}
				if (component.State == HerderPartyComponent.HerderState.Roaming)
				{
					component.State = HerderPartyComponent.HerderState.Herding;
					component.HerdingStartTime = CampaignTime.Now;
				}
				ProduceGoods(component);
				float elapsedHoursUntilNow2 = component.HerdingStartTime.ElapsedHoursUntilNow;
				if (elapsedHoursUntilNow2 >= 6f && MBRandom.RandomFloat < GetEndHerdingChance(elapsedHoursUntilNow2))
				{
					if (MBRandom.RandomFloat < GetEndRoamingChance(component))
					{
						StartReturnToVillage(component);
					}
					else
					{
						GoToNewGrazingPoint(component);
					}
				}
			}
			else if (party.DefaultBehavior == AiBehavior.GoToPoint && (party.Position - party.TargetPosition).LengthSquared < 0.01f)
			{
				party.SetMoveModeHold();
				if (component.State == HerderPartyComponent.HerderState.ReturningToVillage)
				{
					if (ShouldDisbandNow(component))
					{
						DisbandHerder(component);
						return;
					}
					ProcessHerdTurnover(component);
					OnDropOff(component);
					StartRoaming(component);
				}
			}
			else if (party.DefaultBehavior != AiBehavior.GoToPoint)
			{
				Main.Log("[Herder] " + party.StringId + " unrecognised behavior " + party.DefaultBehavior.ToString() + " — resetting to roaming");
				StartRoaming(component);
			}
		}
		catch (Exception ex)
		{
			Main.Log("ERROR in HerderBehavior.OnHourlyTickParty: " + ex.Message);
		}
	}

	private void StartRoaming(HerderPartyComponent component)
	{
		component.State = HerderPartyComponent.HerderState.Roaming;
		component.HerdingStartTime = CampaignTime.Never;
		component.RoamingStartTime = CampaignTime.Now;
		GoToNewGrazingPoint(component);
	}

	private void GoToNewGrazingPoint(HerderPartyComponent component)
	{
		component.State = HerderPartyComponent.HerderState.Roaming;
		component.HerdingStartTime = CampaignTime.Never;
		CampaignVec2 gatePosition = component.Village.Settlement.GatePosition;
		CampaignVec2 campaignVec = gatePosition;
		int num = 20;
		do
		{
			campaignVec = NavigationHelper.FindReachablePointAroundPosition(gatePosition, MobileParty.NavigationType.Default, 12f, 3f);
			num--;
		}
		while (num > 0 && campaignVec.Distance(component.MobileParty.Position) < 3f);
		component.MobileParty.SetMoveGoToPoint(campaignVec, MobileParty.NavigationType.Default);
	}

	private void StartReturnToVillage(HerderPartyComponent component)
	{
		component.State = HerderPartyComponent.HerderState.ReturningToVillage;
		component.HerdingStartTime = CampaignTime.Never;
		CampaignVec2 gatePosition = component.Village.Settlement.GatePosition;
		component.MobileParty.SetMoveGoToPoint(gatePosition, MobileParty.NavigationType.Default);
	}

	private void ProduceGoods(HerderPartyComponent component)
	{
		EnsureItemsCached();
		MobileParty mobileParty = component.MobileParty;
		int animalCount = component.AnimalCount;
		float climateProductionFactor = GetClimateProductionFactor(component.Village);
		switch (component.AnimalType)
		{
		case HerderPartyComponent.HerdType.Cow:
			ProduceItem(mobileParty, _milkItem, animalCount, 0.008f * climateProductionFactor);
			break;
		case HerderPartyComponent.HerdType.Sheep:
			ProduceItem(mobileParty, _milkItem, animalCount, 0.004f * climateProductionFactor);
			ProduceItem(mobileParty, _woolItem, animalCount, 0.002f * climateProductionFactor);
			break;
		case HerderPartyComponent.HerdType.Hog:
			break;
		}
	}

	private void ProduceItem(MobileParty party, ItemObject item, int animalCount, float ratePerAnimal)
	{
		if (item == null)
		{
			return;
		}
		float f = (float)animalCount * ratePerAnimal * MBRandom.RandomFloat;
		int num = MBRandom.RoundRandomized(f);
		if (num > 0)
		{
			float num2 = (float)party.InventoryCapacity - party.TotalWeightCarried;
			int num3 = (int)(num2 / (item.Weight + 0.1f));
			if (num3 < num)
			{
				num = num3;
			}
			if (num > 0)
			{
				party.ItemRoster.AddToCounts(item, num);
			}
		}
	}
