te float GetHerbSeasonalMultiplier(CampaignTime.Seasons season, ClimateZone climate)
	{
		if (climate == ClimateZone.Tropical || climate == ClimateZone.Desert)
		{
			return season switch
			{
				CampaignTime.Seasons.Spring => 1.2f, 
				CampaignTime.Seasons.Summer => 1.4f, 
				CampaignTime.Seasons.Autumn => 1.4f, 
				CampaignTime.Seasons.Winter => 1f, 
				_ => 1f, 
			};
		}
		return season switch
		{
			CampaignTime.Seasons.Spring => 1.5f, 
			CampaignTime.Seasons.Summer => 1.8f, 
			CampaignTime.Seasons.Autumn => 1f, 
			CampaignTime.Seasons.Winter => 0.1f, 
			_ => 1f, 
		};
	}

	private static float GetRanchCountMultiplier(Village village)
	{
		if (_ranchCounts == null || village?.Settlement == null)
		{
			return 1f;
		}
		if (!_ranchCounts.TryGetValue(village.Settlement.StringId, out var value) || value <= 1)
		{
			return 1f;
		}
		HAPSettings? instance = GlobalSettings<HAPSettings>.Instance;
		float num = ((instance != null && instance.EnableAIUpgradeItemConsumption) ? 0.2f : 0.3f);
		return 1f / (float)Math.Pow(value, num);
	}

	private static void ApplyExternalModifiers(Village village, ref ExplainedNumber result)
	{
		List<ExternalTooltipEntry> externalTooltipEntries = GetExternalTooltipEntries(village);
		if (externalTooltipEntries.Count == 0)
		{
			return;
		}
		foreach (ExternalTooltipEntry item in externalTooltipEntries)
		{
			if (item.Factor != 0f)
			{
				result.AddFactor(item.Factor, new TextObject(item.Label));
			}
		}
	}

	public static void RegisterTooltipProvider(ExternalTooltipProvider provider)
	{
		if (provider != null && !_externalProviders.Contains(provider))
		{
			_externalProviders.Add(provider);
			Main.Log($"External tooltip provider registered (total: {_externalProviders.Count})");
		}
	}

	public static List<ExternalTooltipEntry> GetExternalTooltipEntries(Village village)
	{
		List<ExternalTooltipEntry> list = new List<ExternalTooltipEntry>();
		foreach (ExternalTooltipProvider externalProvider in _externalProviders)
		{
			try
			{
				List<ExternalTooltipEntry> list2 = externalProvider(village);
				if (list2 != null)
				{
					list.AddRange(list2);
				}
			}
			catch (Exception ex)
			{
				Main.Log("ERROR in external tooltip provider: " + ex.Message);
			}
		}
		return list;
	}

		}

		{
			_annualProfileCache.Clear();
			_annualProfileCacheYear = num;
		}
		string stringId = village.StringId;
		if (_annualProfileCache.TryGetValue(stringId, out var value))
		{
			return value;
		}

		ClimateZone climateZone = GetClimateZone(village);

					continue;
				}

					{

		}
		if (village.VillageType?.Productions != null)
		{
