	{

		}

		{
			Main.Log("NavalRaidCompletedFix ERROR: " + ex.Message);
		}
	}

	private static PolicyObject ResolvePolicyLazy()
	{
		if (_policyResolved)
		{
			return _raidersSpoils;
		}
		_policyResolved = true;
		try
		{
			Type type = AccessTools.TypeByName("NavalDLC.CampaignBehaviors.NavalPolicies");
			if (type == null)
			{
				return null;
			}
			_raidersSpoils = (AccessTools.Property(type, "RaidersSpoils")?.GetValue(null) as PolicyObject) ?? (AccessTools.Field(type, "RaidersSpoils")?.GetValue(null) as PolicyObject);
		}
		catch (Exception ex)
		{
			Main.Log("NavalRaidCompletedFix: Failed to resolve RaidersSpoils policy: " + ex.Message);
		}
		return _raidersSpoils;
	}

	private static bool Prefix(BattleSideEnum winnerSide, RaidEventComponent raidEvent)
	{
		if (winnerSide != BattleSideEnum.Attacker)
		{
			return false;
		}
		try
		{
			Kingdom kingdom = (raidEvent.AttackerSide.LeaderParty.Owner?.Clan)?.Kingdom;

				return false;
			}
			foreach (MapEventParty party in raidEvent.AttackerSide.Parties)

				if (party.Party.LeaderHero != null)

				}
			}
		}
		catch (Exception ex)

			Main.Log("NavalRaidCompletedFix ERROR: " + ex.Message);
