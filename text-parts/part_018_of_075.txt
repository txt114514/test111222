	private bool IsHerderPartyEncounter(out bool hasGoods, out bool isEnemy)
	{
		hasGoods = false;
		isEnemy = false;
		if (PlayerEncounter.Current == null)
		{
			return false;
		}
		if (Campaign.Current.CurrentConversationContext != ConversationContext.PartyEncounter)
		{
			return false;
		}
		PartyBase encounteredParty = PlayerEncounter.EncounteredParty;
		if (encounteredParty == null || !encounteredParty.IsMobile)
		{
			return false;
		}
		MobileParty mobileParty = encounteredParty.MobileParty;
		if (!HerderPartyComponent.IsHerderParty(mobileParty))
		{
			return false;
		}
		HerderPartyComponent component = HerderPartyComponent.GetComponent(mobileParty);
		if (component == null)
		{

		if (settlement != null)
		{
			MBTextManager.SetTextVariable("SETTLEMENT_NAME", settlement.EncyclopediaLinkWithName);

		TextObject text = ((component.AnimalType == HerderPartyComponent.HerdType.Cow) ? new TextObject("{=HAP_HerderCowText}cattle") : new TextObject("{=HAP_HerderSheepText}sheep"));

	private void herder_accept_bribe_on_consequence()
	{
		PartyBase encounteredParty = PlayerEncounter.EncounteredParty;
		if (encounteredParty == null)
		{
			return;
		}
		int cachedBribeGold = _cachedBribeGold;
		ItemRoster itemRoster = _cachedBribeItems ?? new ItemRoster();
		for (int i = 0; i < itemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = itemRoster.GetElementCopyAtIndex(i);
			if (elementCopyAtIndex.Amount > 0)
			{
				PartyBase.MainParty.ItemRoster.AddToCounts(elementCopyAtIndex.EquipmentElement, elementCopyAtIndex.Amount);
				encounteredParty.ItemRoster.AddToCounts(elementCopyAtIndex.EquipmentElement, -elementCopyAtIndex.Amount);
			}
		}
		if (cachedBribeGold > 0)
		{
			GiveGoldAction.ApplyForPartyToCharacter(encounteredParty, Hero.MainHero, cachedBribeGold);
		}
		BeHostileAction.ApplyMinorCoercionHostileAction(PartyBase.MainParty, encounteredParty);
		Hero hero = encounteredParty.MobileParty?.HomeSettlement?.Village?.Bound?.Owner;
		if (hero != null && hero.Clan != Clan.PlayerClan)
		{
			ChangeRelationAction.ApplyPlayerRelation(hero, -3);
		}
		SkillLevelingManager.OnLoot(MobileParty.MainParty, encounteredParty.MobileParty, itemRoster, attacked: false);
		_cachedBribeGold = 0;
		_cachedBribeItems = null;
		SetHerderInteraction(encounteredParty.MobileParty, 2);
		HerderPartyComponent component = HerderPartyComponent.GetComponent(encounteredParty.MobileParty);
		if (component != null)
		{
			HerderPartyCampaignBehavior.OnAnimalsRemovedFromHerd(component);
		}

	}

	private bool HasValuableGoods(PartyBase party)
	{
		if (party?.ItemRoster == null)
		{
			return false;
		}
		for (int i = 0; i < party.ItemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = party.ItemRoster.GetElementCopyAtIndex(i);
			if (elementCopyAtIndex.EquipmentElement.Item != null && elementCopyAtIndex.Amount > 0 && elementCopyAtIndex.EquipmentElement.Item.ItemCategory != DefaultItemCategories.PackAnimal)
			{
				return true;
			}
		}
		return party.MobileParty != null && party.MobileParty.PartyTradeGold > 0;
	}

	private void CalculateHerderBribeAmount(PartyBase party, out int gold, out ItemRoster items)
	{
		gold = 0;
		items = new ItemRoster();
		if (party?.MobileParty == null)
		{
			return;
		}
		gold = party.MobileParty.PartyTradeGold / 2;
		for (int i = 0; i < party.ItemRoster.Count; i++)
		{
			ItemRosterElement elementCopyAtIndex = party.ItemRoster.GetElementCopyAtIndex(i);
			if (elementCopyAtIndex.EquipmentElement.Item != null && elementCopyAtIndex.EquipmentElement.Item.ItemCategory != DefaultItemCategories.PackAnimal)
			{
				int elementNumber = party.ItemRoster.GetElementNumber(i);
				if (elementNumber > 0)
				{
					int number = MathF.Max(1, elementNumber / 2);
					items.AddToCounts(elementCopyAtIndex.EquipmentElement, number);
				}
			}
		}
	}

	private void SetHerderInteraction(MobileParty party, int interaction)
	{
		if (party != null)
		{
			_interactedHerders[party] = interaction;
		}
	}

	private int GetHerderInteraction(MobileParty party)
	{
		if (party != null && _interactedHerders.TryGetValue(party, out var value))
		{
			return value;
		}
		return 0;
	}
