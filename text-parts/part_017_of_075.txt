	private void OnDropOff(HerderPartyComponent component)
	{
		MobileParty mobileParty = component.MobileParty;
		Village village = component.Village;
		if (mobileParty == null || village == null)
		{
			return;
		}
		string text = ((component.AnimalType == HerderPartyComponent.HerdType.Cow) ? "cow" : ((component.AnimalType == HerderPartyComponent.HerdType.Hog) ? "hog" : "sheep"));
		List<ItemRosterElement> list = new List<ItemRosterElement>();
		for (int num = mobileParty.ItemRoster.Count - 1; num >= 0; num--)
		{
			ItemRosterElement elementCopyAtIndex = mobileParty.ItemRoster.GetElementCopyAtIndex(num);
			ItemObject item = elementCopyAtIndex.EquipmentElement.Item;
			if (item != null && elementCopyAtIndex.Amount > 0 && !(item.StringId == text))
			{
				list.Add(elementCopyAtIndex);
				village.Settlement.ItemRoster.AddToCounts(item, elementCopyAtIndex.Amount);
				mobileParty.ItemRoster.AddToCounts(item, -elementCopyAtIndex.Amount);
				bool flag = false;
			}
		}
		FireSuppliedNotifications(mobileParty, village.Settlement, list);
		DepositTradeGold(mobileParty, village);
		TryReinforceParty(component);
	}

	private void TryReinforceParty(HerderPartyComponent component)
	{
		Village village = component.Village;
		if (village == null || component.Village.Settlement.Party.MapEvent != null || village.VillageState != Village.VillageStates.Normal)
		{
			return;
		}
		PartyTemplateObject partyTemplateObject = component.Village.Settlement.Culture?.VillagerPartyTemplate;
		if (partyTemplateObject == null || partyTemplateObject.Stacks.Count == 0)
		{
			return;
		}
		int num = Math.Min(35, Math.Max(18, (int)(village.Hearth / 40f) + 12));
		int num2 = num - component.MobileParty.MemberRoster.TotalManCount;
		if (num2 > 0)
		{
			if (num2 > (int)village.Hearth)
			{
				num2 = (int)village.Hearth;
			}
			if (num2 > 0)
			{
				village.Hearth -= (num2 + 1) / 2;
				CharacterObject character = partyTemplateObject.Stacks.GetRandomElement().Character;
				component.MobileParty.MemberRoster.AddToCounts(character, num2);
				MilitiaGuardHelper.EnsureMilitiaGuards(component.MobileParty, village);
			}
		}
	}

	private float GetEndHerdingChance(float herdingHours)
	{
		if (herdingHours < 6f)
		{
			return 0f;
		}
		if (herdingHours > 10f)
		{
			return 1f;

		return (herdingHours - 6f) / 4f;
	}

	private float GetEndRoamingChance(HerderPartyComponent component)
	{
		MobileParty mobileParty = component.MobileParty;
		if (mobileParty.TotalWeightCarried >= (float)mobileParty.InventoryCapacity)
		{
			return 1f;
		}
		float elapsedDaysUntilNow = component.RoamingStartTime.ElapsedDaysUntilNow;
		if (elapsedDaysUntilNow < 1f)
		{
			return 0f;
		}
		if (elapsedDaysUntilNow > 3f)
		{
			return 1f;
		}
		int num = 0;
		string text = ((component.AnimalType == HerderPartyComponent.HerdType.Cow) ? "cow" : ((component.AnimalType == HerderPartyComponent.HerdType.Hog) ? "hog" : "sheep"));
		for (int i = 0; i < mobileParty.ItemRoster.Count; i++)
		{
			ItemObject itemAtIndex = mobileParty.ItemRoster.GetItemAtIndex(i);
			if (itemAtIndex != null && itemAtIndex.StringId != text)
			{
				num += mobileParty.ItemRoster.GetElementNumber(i);
			}
		}
		if (num < 3)
		{
			return 0f;
		}
		int num2 = MathF.Round((3f - elapsedDaysUntilNow) / 2f * 4f);
		return 1f / (float)(num2 + 1);
	}

	private int GetAnimalCountInVillage(Village village, string animalId)
	{
		int num = 0;
		ItemRoster itemRoster = village.Owner.ItemRoster;
		for (int i = 0; i < itemRoster.Count; i++)
		{
			ItemObject itemAtIndex = itemRoster.GetItemAtIndex(i);
			if (itemAtIndex != null && itemAtIndex.StringId == animalId)
			{
				num += itemRoster.GetElementNumber(i);
				break;
			}
		}
		return num;
	}

		string text = ((component.AnimalType == HerderPartyComponent.HerdType.Cow) ? "cow" : ((component.AnimalType == HerderPartyComponent.HerdType.Hog) ? "hog" : "sheep"));

	private void FireSuppliedNotifications(MobileParty party, Settlement villageSettlement, List<ItemRosterElement> items)
	{
		if (items == null || items.Count == 0 || villageSettlement == null || party == null)
		{
			return;
		}
		try
		{
			foreach (ItemRosterElement item in items)
			{
				if (item.EquipmentElement.Item != null && item.Amount > 0)
				{
					CampaignEventDispatcher.Instance.OnItemSold(villageSettlement.Party, party.Party, item, item.Amount, villageSettlement);
				}
			}
		}
		catch (Exception ex)
		{
			Main.Log("ERROR in HerderBehavior.FireSuppliedNotifications: " + ex.Message);
		}
	}

	private void DepositTradeGold(MobileParty party, Village village)
	{
		if (party == null || village == null)
		{
			return;
		}
		int partyTradeGold = party.PartyTradeGold;
		if (partyTradeGold <= 0)
		{
			return;
		}
		try
		{
			float villageTaxRatio = Campaign.Current.Models.SettlementTaxModel.GetVillageTaxRatio(village);
			int num = MBRandom.RoundRandomized((float)partyTradeGold * villageTaxRatio);
			if (num > 0)
			{
				village.TradeTaxAccumulated += num;
			}
			int num2 = partyTradeGold - num;
			if (num2 > 0)
			{
				village.ChangeGold(num2);
			}
			party.PartyTradeGold = 0;
			bool flag = false;
		}
		catch (Exception ex)
		{
			Main.Log("ERROR in HerderBehavior.DepositTradeGold: " + ex.Message);
		}
	}

	private float GetClimateProductionFactor(Village village)
	{
		if (village == null)
		{
			return 1f;
		}
		try
		{
			return VillageProductionModel.GetClimateZoneForVillage(village) switch
			{
				VillageProductionModel.ClimateZone.Desert => 0.5f, 
				VillageProductionModel.ClimateZone.Tropical => 0.75f, 
				VillageProductionModel.ClimateZone.Cold => 0.85f, 
				_ => 1f, 
			};
		}
		catch
		{
			return 1f;
		}
	}

	private bool IsSnowCovered(Settlement settlement)
	{
		try
		{
			if (Campaign.Current?.Models?.MapWeatherModel == null)
			{
				return false;
			}
			Campaign.Current.Models.MapWeatherModel.GetSnowAndRainDataForPosition(settlement.GetPosition2D, CampaignTime.Now, out var snowValue, out var _);
			return snowValue > 0.55f;
		}
		catch
		{
			return false;
		}
	}

	private bool ShouldDisbandNow(HerderPartyComponent component)
	{
		Village village = component.Village;
		if (village == null)
		{
			return true;
		}
		if (village.Hearth < 300f)
		{
			return true;
		}
		if (village.Bound?.Town != null && village.Bound.Town.Security < 60f)
		{
			return true;
		}
		if (village.VillageState != Village.VillageStates.Normal)
		{
			return true;
		}
		if (IsSnowCovered(village.Settlement))
		{
			return true;
		}
		if (CampaignTime.Now.GetSeasonOfYear == CampaignTime.Seasons.Winter)
		{
			return true;
		}
		if (component.OriginalClan != village.Settlement.OwnerClan)
		{
			return true;
		}
		if (component.AnimalCount < 20)
		{
			return true;
		}
		return false;
	}
}
